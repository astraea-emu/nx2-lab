cmake_minimum_required(VERSION 3.25)
project(nx2_dynarmic_research LANGUAGES C CXX ASM)

if(NOT DEFINED NX2_PINNED_DYNARMIC_SOURCE)
  message(FATAL_ERROR "Supply the independently verified, exact-revision Dynarmic checkout")
endif()
set(DYNARMIC_TESTS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_FRONTENDS "A64" CACHE STRING "" FORCE)
set(DYNARMIC_USE_BUNDLED_EXTERNALS ON CACHE BOOL "" FORCE)
set(DYNARMIC_USE_PRECOMPILED_HEADERS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
add_subdirectory("${NX2_PINNED_DYNARMIC_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/dynarmic")

set(NX2_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../..")
add_library(nx2_research_oracle STATIC "${NX2_ROOT}/src/cpu.cpp" "${NX2_ROOT}/src/memory.cpp")
target_include_directories(nx2_research_oracle PUBLIC "${NX2_ROOT}/include")
target_compile_features(nx2_research_oracle PUBLIC cxx_std_23)
add_executable(nx2_dynarmic_owned_trace owned_trace.cpp)
target_compile_features(nx2_dynarmic_owned_trace PRIVATE cxx_std_23)
target_link_libraries(nx2_dynarmic_owned_trace PRIVATE nx2_research_oracle dynarmic)
