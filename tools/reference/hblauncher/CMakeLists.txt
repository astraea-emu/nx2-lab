cmake_minimum_required(VERSION 3.25)
project(nx2_hbx_original_guest LANGUAGES C CXX ASM)

if(NOT DEFINED NX2_PINNED_DYNARMIC_SOURCE)
  message(FATAL_ERROR "Provide frozen and independently checked Dynarmic source")
endif()
set(DYNARMIC_TESTS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_FRONTENDS "A64" CACHE STRING "" FORCE)
set(DYNARMIC_USE_BUNDLED_EXTERNALS ON CACHE BOOL "" FORCE)
set(DYNARMIC_USE_PRECOMPILED_HEADERS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
add_subdirectory("${NX2_PINNED_DYNARMIC_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/dynarmic")

add_executable(nx2_hbx_stage1 original_stage1.cpp)
target_compile_features(nx2_hbx_stage1 PRIVATE cxx_std_23)
target_link_libraries(nx2_hbx_stage1 PRIVATE dynarmic)
if(MSVC)
  target_compile_options(nx2_hbx_stage1 PRIVATE /W4 /WX)
else()
  target_compile_options(nx2_hbx_stage1 PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
